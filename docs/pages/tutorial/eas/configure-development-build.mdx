// App.tsx
import React, { useMemo, useState } from 'react';
import {
  SafeAreaView, View, Text, TextInput, FlatList, Image, TouchableOpacity, Dimensions, StatusBar,
} from 'react-native';

type Estado = 'Diagnóstico' | 'Execução' | 'Concluído';
type Veiculo = {
  id: string;
  matricula: string;
  modelo: string;
  estado: Estado;
  valor: number;
  promessaData: string; // ex "26/05/2023" ou "hoje"
  promessaHora: string; // ex "18:00"
  imagem?: string;
  badge?: 'NOVO' | 'NA' | null;
};

const C = {
  bg: '#F3F6FA',
  card: '#FFFFFF',
  text: '#101828',
  muted: '#6B7280',
  border: '#E5E7EB',
  azulValor: '#0EA5E9',
  barraDiag: '#E85D75',
  barraExec: '#2CBFAE',
  barraConc: '#6ABF69',
  tabOn: '#16A34A',
};

const MOCK: Veiculo[] = [
  { id: '1', matricula: '66-15-EA', modelo: 'RANGE ROVER I', estado: 'Diagnóstico', valor: 655.5, promessaData: '26/05/2023', promessaHora: '18:00', badge: 'NA' },
  { id: '2', matricula: 'PM-49-90', modelo: 'XJ6 4.2 Series 2', estado: 'Diagnóstico', valor: 3525.3, promessaData: '30/05/2023', promessaHora: '18:00', badge: 'NOVO' },
  { id: '3', matricula: '16-OZ-85', modelo: 'Classe E Carrinha (S212)', estado: 'Execução', valor: 1064.58, promessaData: 'hoje', promessaHora: '18:00' },
  { id: '4', matricula: '93-OB-96', modelo: 'YARIS (_P13_) 1.5 Hybrid', estado: 'Diagnóstico', valor: 0, promessaData: 'hoje', promessaHora: '18:00' },
  { id: '5', matricula: '30-XI-36', modelo: 'ForTwo Coupé (453)', estado: 'Execução', valor: 26.36, promessaData: '09/06/2023', promessaHora: '18:00' },
  { id: '6', matricula: '66-15-EA', modelo: 'RANGE ROVER I', estado: 'Execução', valor: 12300.88, promessaData: '30/06/2023', promessaHora: '17:00' },
];

const estadoBarra = (e: Estado) => e === 'Diagnóstico' ? C.barraDiag : e === 'Execução' ? C.barraExec : C.barraConc;

const w = Dimensions.get('window').width;
const GAP = 12;
const COL = (w - 16*2 - GAP) / 2;

export default function App() {
  const [q, setQ] = useState('');
  const [tab, setTab] = useState<'Todos' | Estado>('Todos');

  const data = useMemo(() => {
    const t = q.trim().toLowerCase();
    return MOCK.filter(v => {
      const mt = !t || v.matricula.toLowerCase().includes(t) || v.modelo.toLowerCase().includes(t);
      const tt = tab === 'Todos' ? true : v.estado === tab;
      return mt && tt;
    });
  }, [q, tab]);

  const Card = ({ v }: { v: Veiculo }) => (
    <TouchableOpacity
      activeOpacity={0.85}
      style={{
        width: COL,
        backgroundColor: C.card,
        borderRadius: 16,
        padding: 12,
        borderWidth: 1,
        borderColor: C.border,
        marginBottom: GAP,
        shadowColor: '#000', shadowOpacity: 0.06, shadowRadius: 8, shadowOffset: { width: 0, height: 2 },
      }}
      onPress={() => console.log('Abrir detalhes', v.matricula)}
    >
      {v.badge && (
        <View style={{ position: 'absolute', top: -8, left: -8, backgroundColor: '#111827', paddingHorizontal: 8, paddingVertical: 4, borderRadius: 999 }}>
          <Text style={{ color: '#fff', fontSize: 10, fontWeight: '700' }}>{v.badge}</Text>
        </View>
      )}

      <Text style={{ fontWeight: '800', color: C.text, textAlign: 'center' }}>{v.matricula}</Text>
      <Text numberOfLines={1} style={{ color: C.muted, textAlign: 'center', marginBottom: 6 }}>{v.modelo}</Text>

      <View style={{ height: 72, justifyContent: 'center', alignItems: 'center', marginBottom: 8 }}>
        <Image
          source={{ uri: v.imagem || 'https://dummyimage.com/180x80/ffffff/aaa&text=VE%C3%8DCULO' }}
          style={{ width: '100%', height: '100%', resizeMode: 'contain' }}
        />
        <View style={{ position: 'absolute', right: -4, bottom: -4, backgroundColor: '#fff', borderRadius: 20, padding: 6, borderWidth: 1, borderColor: C.border }}>
          <Text style={{ fontSize: 12 }}>ℹ️</Text>
        </View>
      </View>

      <View style={{ height: 6, borderRadius: 4, backgroundColor: '#E5E7EB', overflow: 'hidden' }}>
        <View style={{ width: '100%', height: '100%', backgroundColor: estadoBarra(v.estado) }} />
      </View>

      <View style={{ flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginTop: 6 }}>
        <Text style={{ color: C.muted }}>{v.estado === 'Execução' ? 'Execução Serviços' : v.estado}</Text>
        <Text style={{ color: C.azulValor, fontWeight: '800' }}>
          {v.valor.toLocaleString('pt-PT', { style: 'currency', currency: 'EUR' })}
        </Text>
      </View>

      <View style={{ flexDirection: 'row', justifyContent: 'space-between', marginTop: 8 }}>
        <View style={{ flexDirection: 'row', alignItems: 'center' }}>
          <Text style={{ marginRight: 6 }}>📅</Text>
          <Text style={{ color: C.muted }}>{v.promessaData}</Text>
        </View>
        <Text style={{ color: C.muted }}>{v.promessaHora}</Text>
      </View>
    </TouchableOpacity>
  );

  const Chip = ({ label }: { label: 'Todos' | Estado }) => {
    const on = tab === label;
    return (
      <TouchableOpacity
        onPress={() => setTab(label)}
        style={{
          marginRight: 8, borderRadius: 999, borderWidth: 1,
          borderColor: on ? C.tabOn : C.border,
          backgroundColor: on ? '#EAFBF0' : '#fff',
          paddingVertical: 8, paddingHorizontal: 12,
        }}
      >
        <Text style={{ color: on ? C.tabOn : C.muted, fontWeight: on ? '700' : '500' }}>{label}</Text>
      </TouchableOpacity>
    );
  };

  return (
    <SafeAreaView style={{ flex: 1, backgroundColor: C.bg }}>
      <StatusBar barStyle="dark-content" />
      <View style={{ paddingHorizontal: 16, paddingTop: 8, paddingBottom: 10 }}>
        <View style={{ flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between', marginBottom: 8 }}>
          <Text style={{ fontSize: 18, fontWeight: '800', color: C.text }}>Impacto Car</Text>
          <View style={{
            width: 32, height: 32, borderRadius: 16, backgroundColor: '#E2E8F0',
            alignItems: 'center', justifyContent: 'center',
          }}>
            <Text style={{ fontWeight: '700', color: '#334155' }}>IC</Text>
          </View>
        </View>

        <View style={{
          flexDirection: 'row', alignItems: 'center',
          backgroundColor: '#fff', borderRadius: 12, borderWidth: 1, borderColor: C.border, paddingHorizontal: 10,
        }}>
          <Text style={{ marginRight: 8 }}>🔍</Text>
          <TextInput
            style={{ flex: 1, paddingVertical: 10 }}
            placeholder="Pesquisar"
            placeholderTextColor={C.muted}
            value={q}
            onChangeText={setQ}
          />
        </View>

        <View style={{ flexDirection: 'row', marginTop: 10 }}>
          <Chip label="Todos" />
          <Chip label="Diagnóstico" />
          <Chip label="Execução" />
          <Chip label="Concluído" />
        </View>
      </View>

      <FlatList
        data={data}
        keyExtractor={(i) => i.id}
        numColumns={2}
        columnWrapperStyle={{ paddingHorizontal: 16, justifyContent: 'space-between' }}
        contentContainerStyle={{ paddingBottom: 24 }}
        renderItem={({ item }) => <Card v={item} />}
        ListEmptyComponent={<Text style={{ textAlign: 'center', color: C.muted, marginTop: 40 }}>Nenhum veículo.</Text>}
      />
    </SafeAreaView>
  );
}
